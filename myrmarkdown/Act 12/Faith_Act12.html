x1 <- mouse_lm$AverageDistance
x2 <- mouse_lm$Effort
x3 <- mouse_lm$HabitatCode
y <- mouse_lm$Detected

mouse_lmdf <- data.frame(x1,x2,x3,y)
model <- glm(y~ x1+ x2 + x3,
             data = mouse_lmdf,
             family = binomial)

mouse_lmdf$Probability <- predict(model, type = "response")
mouse_lmdf$Prediction <- ifelse(mouse_lmdf$Probability >= 0.5, 1, 0)

5. After fitting the model, create a logistic regression curve plot for each independent variable used. Export the plots as PDF files.

Average Distance vs. Probability of Detection

finalx <- data.frame(
  x1 = seq(min(mouse_lmdf$x1), max(mouse_lmdf$x1), length.out = 19),
  x2 = mean(mouse_lmdf$x2),
  x3 = levels(mouse_lmdf$x3)[1])

finalx$Probability <- predict(model, newdata = finalx, type = "response")

plot(mouse_lmdf$x1, mouse_lmdf$y,
     pch = 19,
     xlab = "Average Distance",
     ylab = "Probability",
     ylim = c(0,1))

lines(finalx$x1,
      finalx$Probability,
      col = "blue",
      lwd = 3)
